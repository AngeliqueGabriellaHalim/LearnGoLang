---------
STRUCTS 
----------
1. to group many related value with diff types

example:
type line struct{
    x1 int
    y1 int
    x2 int
    y2 int
}

INSTANIATE STRUCT:
1. init all field, follow order of all fields
l1 := line{2, 3, 4, 6}
l2 := line{3, 3, 8, 8}

2. init any order, un-init fields = type's zero value

l1 := line{
x1: 2,
y1: 3,
}

3. init everything with zero value
l1 := line{}

to modify or acces, use dot 

note: doesnt support inheritance


------------
POINTER
-----------
var number *int
var name *string

1. DEFAULT value pointer = nil

2. ambil nilai pointer : &variable --> referencing
3. ambil value var nya : *variable --> dereferencing
4. ubah isi variabel poitner dengan nilai pointer sama = isinya ikutn berubah
5. pointer sebagai parameters

func change(original *int, value int) {
    *original = value
}
func main() {
    var number = 4
    fmt.Println("before :", number) // 4

    change(&number, 10)
    fmt.Println("after  :", number) // 10
}

func (s student) getNameAt(i int) string {
    return strings.Split(s.name, " ")[i-1]
}

METHOD POINTER

----------------------
ANY / INTERFACE KOSONG / interface{}
-----------------------
1. tipe data yg bisa menampung segala jenis data

    var secret interface{}

    secret = "ethan hunt"
    fmt.Println(secret)

    secret = []string{"apple", "mango", "banana"}
    fmt.Println(secret)

    secret = 12.4
    fmt.Println(secret)

2. any merupakan alias interface{}
    var data map[string]any

    data = map[string]any{
        "name":      "ethan hunt",
        "grade":     2,
        "breakfast": []string{"apple", "mango", "banana"},
    }

3. castingnya sama seperti cast interface
    >contoh int
    secret = 2
    var number = secret.(int) * 10
    fmt.Println(secret, "multiplied by 10 is :", number)

    >contoh string
      secret = []string{"apple", "mango", "banana"}
    var fruits = strings.Join(secret.([]string), ", ")
    fmt.Println(fruits, "is my favorite fruits")

    >contoh pointer
    type person struct {
        name string
        age  int
    }

    var secret interface{} = &person{name: "wick", age: 27}
    var name = secret.(*person).name
    fmt.Println(name)
